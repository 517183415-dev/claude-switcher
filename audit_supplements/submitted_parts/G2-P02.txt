000600 | pub fn normalize_image(bytes: &[u8], mime: &str) -> Result<(Vec<u8>, ImageAsset), &'static str> {
000601 |     let _guard = IMAGE_DECODE
000602 |         .lock()
000603 |         .map_err(|_| "image_decoder_unavailable")?;
000604 |     let decoded = decode_image(bytes, mime)?;
000605 |     let (width, height) = decoded.dimensions();
000606 |     let format = if mime == "image/png" {
000607 |         ImageFormat::Png
000608 |     } else {
000609 |         ImageFormat::Jpeg
000610 |     };
000611 |     let pixels = if format == ImageFormat::Png {
000612 |         DynamicImage::ImageRgba8(decoded.into_rgba8())
000613 |     } else {
000614 |         DynamicImage::ImageRgb8(decoded.into_rgb8())
000615 |     };
000616 |     let mut normalized = Vec::new();
000617 |     pixels
000618 |         .write_to(&mut Cursor::new(&mut normalized), format)
000619 |         .map_err(|_| "image_encode")?;
000620 |     if normalized.len() > MAX_IMAGE_BYTES {
000621 |         return Err("image_size");
000622 |     }

000624 |         sha256: signed::digest(&normalized),
000625 |         mime: mime.into(),
000626 |         bytes: normalized.len() as u32,
000627 |         width,
000628 |         height,
000629 |     };
000630 |     Ok((normalized, asset))
000631 | }
