000605 | impl StopSupervisor {
000606 |     fn new(

000608 |         timeout: Duration,
000609 |         retry: Duration,
000610 |     ) -> Self {
000611 |         Self {

000613 |             job: None,
000614 |             pending: None,
000615 |             confirmed_epoch: None,
000616 |             failed: false,
000617 |             timeout,
000618 |             retry,
000619 |             retry_at: tokio::time::Instant::now(),
000620 |         }
000621 |     }
000622 |     fn request(&mut self, epoch: u64, request: StopRequest) {
000623 |         self.confirmed_epoch = None;
000624 |         if self.job.as_ref().is_some_and(|job| job.epoch == epoch) {
000625 |             return;
000626 |         }
000627 |         self.pending = Some((epoch, request));
000628 |     }
000629 |     fn start_ready(&mut self) {
000630 |         if self.job.is_some() || tokio::time::Instant::now() < self.retry_at {
000631 |             return;
000632 |         }
000633 |         if let Some((epoch, request)) = self.pending.take() {
000634 |             let consumer = self.consumer.clone();
000635 |             let operation = request.clone();

000637 |                 epoch,
000638 |                 request,
000639 |                 deadline: tokio::time::Instant::now() + self.timeout,
000640 |                 timed_out: false,
000641 |                 handle: tokio::task::spawn_blocking(move || consumer(&operation)),
000642 |             });
000643 |         }
000644 |     }
000645 |     async fn join(&mut self) -> StopCompletion {
000646 |         let result = (&mut self.job.as_mut().expect("active stop slot").handle)

000648 |             .unwrap_or(Err("guardian_stop_worker_join"));

000652 |             request: job.request,
000653 |             timed_out: job.timed_out || tokio::time::Instant::now() >= job.deadline,
000654 |             result,
000655 |         }
000656 |     }
000657 |     fn finish(&mut self, completion: StopCompletion, current_epoch: u64) -> bool {
000658 |         if !completion.timed_out
000659 |             && completion.result.is_ok()
000660 |             && completion.epoch == current_epoch
000661 |             && self.pending.is_none()

000663 |             self.failed = false;
000664 |             self.confirmed_epoch = Some(current_epoch);
000665 |             true
000666 |         } else {
000667 |             self.failed = true;
000668 |             self.confirmed_epoch = None;
000669 |             // A late success from a timed-out or old epoch cannot acknowledge a newer request.
000670 |             if self.pending.is_none() && completion.epoch == current_epoch {
000671 |                 self.pending = Some((current_epoch, completion.request));
000672 |             }
000673 |             self.retry_at = tokio::time::Instant::now() + self.retry;
000674 |             false
000675 |         }
000676 |     }
