000990 |         &self,

000992 |         lock: &LifecycleLock,
000993 |         stopped: &dyn Fn() -> io::Result<()>,
000994 |     ) -> io::Result<()> {
000995 |         self.check_lock(lock)?;

000998 |             return Err(fail("用户修复事务尚未完整验证"));

001030 |     fn restore_inner(
001031 |         &self,
001032 |         journal: &Journal,
001033 |         lock: &LifecycleLock,
001034 |         stopped: &dyn Fn() -> io::Result<()>,
001035 |         prefs: &dyn PreferenceWriter,
001036 |     ) -> io::Result<()> {
001037 |         self.check_lock(lock)?;
001038 |         if journal.rolled_back {
001039 |             return self.finalize_record(journal, lock);
001040 |         }
001041 |         stopped()?;
001042 |         if journal.committed {
001043 |             return Err(fail("用户修复已提交，不允许按未提交事务回滚"));
001044 |         }
001045 |         for change in &journal.changes {
001046 |             self.check_user_path(&self.path(change.kind))?;
001047 |         }
001048 |         let backup = self.directory().join(&journal.id);
001049 |         check_directory(&backup, self.owner)?;
001050 |         // Validate every source before restoring any source. Unknown changes are never overwritten.
001051 |         for (index, change) in journal.changes.iter().take(journal.attempted).enumerate() {
001052 |             let (current, bytes) = snapshot(&self.path(change.kind), self.owner)?;
001053 |             if original_image(&current, &change.before) {

001055 |             }

001059 |                     if matches!(change.kind, Kind::Proxy | Kind::PinnedRoutes)
001060 |                         || !preferences::same_key(
001061 |                             original.as_deref(),
001062 |                             bytes.as_deref(),
001063 |                             change.kind,
001064 |                         )?
001065 |                     {
001066 |                         return Err(fail("用户配置已被外部修改，拒绝覆盖恢复"));
