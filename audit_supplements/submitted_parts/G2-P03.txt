000580 | async fn collect_reports(
000581 |     mut tasks: tokio::task::JoinSet<CheckResult>,
000582 |     deadline: Option<tokio::time::Instant>,
000583 |     observed_at: u64,
000584 |     generation: &str,
000585 | ) -> Vec<CheckResult> {
000586 |     let mut checks = Vec::new();
000587 |     while !tasks.is_empty() {
000588 |         let next = match deadline {
000589 |             Some(deadline) => tokio::time::timeout_at(deadline, tasks.join_next()).await,
000590 |             None => Ok(tasks.join_next().await),
000591 |         };
000592 |         match next {
000593 |             Ok(Some(Ok(row))) => checks.push(row),
000594 |             Ok(Some(Err(_))) => {}
000595 |             _ => {
000596 |                 tasks.abort_all();
000597 |                 break;
000598 |             }
000599 |         }
000600 |     }
000601 |     for id in CheckId::ALL {
000602 |         if !checks.iter().any(|r| r.id == id) {
000603 |             checks.push(CheckResult {
000604 |                 id,
000605 |                 status: CheckStatus::Timeout,
000606 |                 reason: "检查到达总截止时间或任务未完成".into(),
000607 |                 observed_at,
000608 |                 duration_ms: 30_000,
000609 |                 valid_until: observed_at + EVIDENCE_TTL_SECONDS,
000610 |                 network_generation: generation.into(),
000611 |                 evidence: vec![],
000612 |             });
000613 |         }
000614 |     }
000615 |     checks.sort_by_key(|r| CheckId::ALL.iter().position(|id| *id == r.id));
