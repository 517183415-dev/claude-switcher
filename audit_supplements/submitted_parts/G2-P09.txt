000989 |     fn recover_runtime_with(
000990 |         &self,
000991 |         lock: &storage::LifecycleLock,
000992 |         status: impl FnOnce(String) -> Result<HelperResponse, &'static str>,
000993 |         runtime: &mut dyn Runtime,
000994 |     ) -> Result<RecoveryOutcome, &'static str> {
000995 |         if !self.has_region_journal()? {
000996 |             let user_path = self
000997 |                 .home
000998 |                 .join(".claude-switcher/v2/user-repairs/journal.json");
000999 |             match fs::symlink_metadata(&user_path) {
001000 |                 Ok(_) => {
001001 |                     // Do not let the user coordinator's existence check hide an unsafe marker.
001002 |                     storage::read_json::<serde_json::Value>(&user_path)
001003 |                         .map_err(|_| "用户配置修复记录无法安全读取；请保留事务与备份")?;
001004 |                 }
001005 |                 Err(error) if error.kind() == std::io::ErrorKind::NotFound => {}
001006 |                 Err(_) => return Err("无法读取用户配置修复状态"),
001007 |             }
001008 |             return UserRepairCoordinator::for_current_user(self.edition)
001009 |                 .map_err(|_| "无法确认用户修复目录")?
001010 |                 .recover_with_lock(lock)
001011 |                 .map(|recovered| {
001012 |                     if recovered {
001013 |                         RecoveryOutcome::Restored
001014 |                     } else {
001015 |                         RecoveryOutcome::NoWork
001016 |                     }
001017 |                 })
001018 |                 .map_err(|_| "用户配置恢复未完成；请保留事务与备份并核对冲突");
001019 |         }
001020 |         let mut journal = self.read_journal()?;
001021 |         let user = self.user_coordinator(journal.user_required)?;
001022 |         if journal.stage == RepairStage::UserPending {
001023 |             self.restore_runtime(&mut journal, user.as_ref(), lock, runtime)?;
001024 |             return Ok(RecoveryOutcome::Restored);
001025 |         }
001026 |         let response = status(journal.plan.id.clone())
001027 |             .map_err(|_| "系统组件仍在执行或结果未知；已保留事务，请稍后重试恢复")?;
001028 |         match resolution(&journal, &response) {
001029 |             Resolution::Applied => {
001030 |                 self.commit_runtime(&mut journal, user.as_ref(), lock, runtime)?;
001031 |                 Ok(RecoveryOutcome::Committed)
001032 |             }
001033 |             Resolution::Restored => {
001034 |                 self.restore_runtime(&mut journal, user.as_ref(), lock, runtime)?;
001035 |                 Ok(RecoveryOutcome::Restored)
001036 |             }
001037 |             Resolution::Unknown => Err("系统修复尚未得到确定结果；已保留全部事务与备份"),
001038 |         }
001039 |     }
001040 | }
